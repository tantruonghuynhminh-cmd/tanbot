/* ========= BƯỚC 3: BUILD CANDIDATES & KIỂM TRA ĐỒNG THUẬN 24H x 1H ========= */ const candidates = []; for (const t of top120Tickers) { const inst = t.instId; const last = Number(t.last); const delta24h = t.delta24h || 0; const dataInfo = atrPriceMap[inst] || {}; const { atr, price1h, candles, candles1h, candles4h, volAvg1h, rsi, rsi14_1h, vwapDistPct, bbZ20, completedVol, priorVolAvg, volSpread, // ===== BB 1H + ATR ===== upperBB1h, lowerBB1h, bbMid1h, bbStd1h, longExcessATR1h, shortExcessATR1h } = dataInfo; // ========================================================= // CORE DATA CHECK // ========================================================= // SCAN/RANKING chỉ cần dữ liệu lõi; // feature Entry thiếu không được làm mất candidate. if ( !last || !price1h || !Number.isFinite(Number(atr)) || Number(atr) <= 0 ) { console.warn('[SCAN CORE DROP]', { inst, last, price1h, atr }); continue; } // ========================================================= // ENTRY FEATURE WARNING // ========================================================= if ( !isFiniteValue(rsi14_1h) || !isFiniteValue(vwapDistPct) || !isFiniteValue(bbZ20) ) { console.warn( '[ENTRY FEATURE MISSING — CANDIDATE KEPT]', { inst, rsi14_1h, vwapDistPct, bbZ20 } ); } // ========================================================= // DELTA 1H THỰC TẾ // ========================================================= const delta = (last - price1h) / price1h * 100; // ========================================================= // DELTA 1H QUYẾT ĐỊNH HƯỚNG LỆNH // ========================================================= // > 0 = LONG // < 0 = SHORT const side = delta > 0 ? 'buy' : 'sell'; let tradeCase = side === 'buy' ? "Uptrend Tiếp Diễn 📈" : "Downtrend Tiếp Diễn 📉"; let isSpecial = false; let isWatching = false; // ========================================================= // VOLUME // ========================================================= const vol24h = parseFloat(t.volCcy24h || 0); const currentVol = isFiniteValue(completedVol) ? Number(completedVol) : null; // ========================================================= // KIỂM TRA XU HƯỚNG ĐA KHUNG (GIỮ NGUYÊN) // ========================================================= let trendUp = false; let trendDown = false; // ========================================================= // BB & RSI SPECIAL CASES // ========================================================= let upperBB = Number.isFinite(Number(upperBB1h)) ? Number(upperBB1h) : null; let lowerBB = Number.isFinite(Number(lowerBB1h)) ? Number(lowerBB1h) : null; // ========================================================= // BB 1H + ATR EXTREME FILTER // ========================================================= /* * Mục đích: * * LONG: * Giá vượt Upper BB quá xa * => không mua đuổi đỉnh. * * SHORT: * Giá vượt Lower BB quá xa * => không bán đuổi đáy. * * Ngưỡng: * > 0.5 ATR = BLOCK * * Ví dụ: * last = UpperBB + 0.7 ATR * => BLOCK LONG * * last = LowerBB - 0.8 ATR * => BLOCK SHORT * * Nếu BB không có dữ liệu: * => KHÔNG block candidate. */ let bbExcessATR = 0; let bbExtreme = false; if ( Number.isFinite(Number(upperBB1h)) && Number.isFinite(Number(lowerBB1h)) && Number.isFinite(Number(atr)) && Number(atr) > 0 ) { if (side === 'buy') { // Giá vượt Upper BB bao nhiêu ATR const excessLong = Number.isFinite(Number(longExcessATR1h)) ? Number(longExcessATR1h) : (last - Number(upperBB1h)) / Number(atr); bbExcessATR = excessLong; // Chỉ block khi thực sự vượt Upper // và vượt quá 0.5 ATR if (excessLong > 0.5) { bbExtreme = true; console.warn( '[BB1H ATR BLOCK LONG]', { inst, last, upperBB1h, atr, excessATR: excessLong, delta } ); continue; } } else { // Giá vượt Lower BB bao nhiêu ATR const excessShort = Number.isFinite(Number(shortExcessATR1h)) ? Number(shortExcessATR1h) : (Number(lowerBB1h) - last) / Number(atr); bbExcessATR = excessShort; // Chỉ block khi thực sự vượt Lower // và vượt quá 0.5 ATR if (excessShort > 0.5) { bbExtreme = true; console.warn( '[BB1H ATR BLOCK SHORT]', { inst, last, lowerBB1h, atr, excessATR: excessShort, delta } ); continue; } } } // ========================================================= // TP / SL // ========================================================= const { tp, sl } = calcTP_SL( last, atr, side === 'buy' ); // ========================================================= // BUILD CANDIDATE // ========================================================= candidates.push({ inst, instId: inst, // PRICE last, price1h, // MOMENTUM delta, delta24h, // ATR atr, // TP / SL tp, sl, // SIDE side, tradeCase, // VOLUME vol24h, currentVol, volSpread, // SPECIAL isSpecial, // BB 1H upperBB, lowerBB, // BB middle/std nếu cần debug hoặc ranking sau này bbMid1h: Number.isFinite(Number(bbMid1h)) ? Number(bbMid1h) : null, bbStd1h: Number.isFinite(Number(bbStd1h)) ? Number(bbStd1h) : null, // Khoảng cách vượt BB tính theo ATR bbExcessATR: Number.isFinite(Number(bbExcessATR)) ? Number(bbExcessATR) : 0, bbExtreme, // ENTRY FEATURES rsi14_1h: isFiniteValue(rsi14_1h) ? Number(rsi14_1h) : null, vwapDistPct: isFiniteValue(vwapDistPct) ? Number(vwapDistPct) : null, bbZ20: isFiniteValue(bbZ20) ? Number(bbZ20) : null, completedVol: isFiniteValue(completedVol) ? Number(completedVol) : null, priorVolAvg: isFiniteValue(priorVolAvg) ? Number(priorVolAvg) : null, volSpread: isFiniteValue(volSpread) ? Number(volSpread) : null, isWatching }); }
